% ------------------------------------------------------------------ 6 -----
\begin{frame}[t,plain]
  \Chrome{05 \textperiodcentered\ THE MISTAKE}{A plausible default was\\worse than zero.}{}{06}
  \FigureAt{0.770}{2.795}{8.26}{\FigMistake}
  \note{The first null-prior model used a plausible global default and scored 49.64 MAE. Zero-fill scored 21.34 and the conditional model 20.48 at 85.3\% coverage. A feature is present in only 4.1\% of null cases, so the default was worse than assuming nothing.}
\end{frame}

% ------------------------------------------------------------------ 7 -----
\begin{frame}[t,plain]

  \FigureAt{0.878}{2.332}{8.33}{\FigCodebase}
  \note{The same feature means different things in different repos, so the model refuses to discount. It reads churn, fan-in, complexity, coverage and authorship from 29,059 real commits. PR lead time was rejected as a signal: beta of -0.892. Signals can only widen the interval.}
\end{frame}
