  \pgfmathsetmacro{\FigH}{7.2*#1}%
  \begin{tikzpicture}[remember picture,overlay]
    \node[anchor=north,inner sep=0pt,outer sep=0pt]
      at ([xshift=#3cm,yshift=-#2cm]current page.north west)
      {\raisebox{0pt}[\FigH cm][0pt]{\scalebox{#1}{#4}}};
  \end{tikzpicture}}
